    if wallet["plan_tier"] != "agency" and wallet["credits_remaining"] < cost:
        raise HTTPException(

        )
    if wallet["plan_tier"] != "agency":
        wallet["credits_remaining"] -= cost
    wallet["total_generations"] += 1
    return wallet
